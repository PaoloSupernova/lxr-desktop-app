cmake_minimum_required(VERSION 3.22)
project(PulseFoundry VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(MSVC)
  add_compile_options(/utf-8 /bigobj)
else()
  add_compile_options(-fno-strict-aliasing -fwrapv)
endif()
option(PF_BUILD_PLUGIN "Build standalone and VST3" ON)
option(PF_BUILD_TESTS "Build headless tests" ON)
option(PF_ASIO "Enable Windows ASIO (JUCE bundled SDK)" ON)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
file(GLOB_RECURSE FIRMWARE_INPUTS CONFIGURE_DEPENDS ../Core/Inc/*.h ../Core/Src/Soundengine/*.c ../Core/Src/Preset/initKit.c)
add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/generated/Firmware.h ${CMAKE_CURRENT_BINARY_DIR}/generated/Parameters.h ${CMAKE_CURRENT_BINARY_DIR}/generated/ParameterSpecs.inc
  COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/port_firmware.py ${CMAKE_CURRENT_BINARY_DIR}/generated/Firmware.h
  DEPENDS tools/port_firmware.py ${FIRMWARE_INPUTS} VERBATIM)
add_library(pf_engine src/Engine.cpp ${CMAKE_CURRENT_BINARY_DIR}/generated/Firmware.h)
target_include_directories(pf_engine PUBLIC src ${CMAKE_CURRENT_BINARY_DIR}/generated)
if(PF_BUILD_TESTS)
  enable_testing()
  add_executable(pf_tests tests/EngineTests.cpp)
  target_link_libraries(pf_tests PRIVATE pf_engine)
  add_test(NAME native_engine COMMAND pf_tests)
  add_executable(pf_parameter_stress tests/ParameterStress.cpp)
  target_link_libraries(pf_parameter_stress PRIVATE pf_engine)
  add_test(NAME sound_parameter_boundaries COMMAND pf_parameter_stress)
  add_executable(pf_render tools/Render.cpp)
  target_link_libraries(pf_render PRIVATE pf_engine)
endif()
if(PF_BUILD_PLUGIN)
  set(PF_JUCE_DIR "" CACHE PATH "Local JUCE 7.0.12 source checkout")
  if(NOT PF_JUCE_DIR AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/JUCE/CMakeLists.txt")
    set(PF_JUCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/JUCE")
  endif()
  if(PF_JUCE_DIR)
    add_subdirectory(${PF_JUCE_DIR} juce)
  else()
    include(FetchContent)
    FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 4f43011b96eb0636104cb3e433894cda98243626)
    FetchContent_MakeAvailable(JUCE)
  endif()
  if(WIN32 AND PF_ASIO)
    # GPLv3 version of the ASIO SDK, kept separately from JUCE 7.
    include_directories(${CMAKE_CURRENT_SOURCE_DIR}/third_party/asio)
  endif()
  set(PF_AUTO_MANIFEST TRUE)
  if(CMAKE_CROSSCOMPILING)
    set(PF_AUTO_MANIFEST FALSE)
  endif()
  juce_add_plugin(PulseFoundry
    COMPANY_NAME "Paul Holliday" BUNDLE_ID "com.paulholliday.pulsefoundry"
    PLUGIN_MANUFACTURER_CODE PlHo PLUGIN_CODE PuFo
    FORMATS Standalone VST3 PRODUCT_NAME "Pulse Foundry"
    IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE
    COPY_PLUGIN_AFTER_BUILD FALSE VST3_AUTO_MANIFEST ${PF_AUTO_MANIFEST} VST3_CATEGORIES Instrument Drum)
  target_sources(PulseFoundry PRIVATE src/Processor.cpp src/Editor.cpp)
  target_link_libraries(PulseFoundry PRIVATE pf_engine juce::juce_audio_utils juce::juce_audio_processors juce::juce_gui_extra)
  target_compile_definitions(PulseFoundry PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0
    JUCE_DISPLAY_SPLASH_SCREEN=0 JUCE_REPORT_APP_USAGE=0 JUCE_ASIO=$<AND:$<BOOL:${WIN32}>,$<BOOL:${PF_ASIO}>>)
  if(PF_BUILD_TESTS)
    add_executable(pf_plugin_tests tests/PluginTests.cpp)
    target_link_libraries(pf_plugin_tests PRIVATE PulseFoundry)
    target_include_directories(pf_plugin_tests PRIVATE $<TARGET_PROPERTY:PulseFoundry,INCLUDE_DIRECTORIES>)
    target_compile_definitions(pf_plugin_tests PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
    add_test(NAME plugin_state_and_routing COMMAND pf_plugin_tests)
    add_executable(pf_preview tools/Preview.cpp)
    target_link_libraries(pf_preview PRIVATE PulseFoundry)
    target_include_directories(pf_preview PRIVATE $<TARGET_PROPERTY:PulseFoundry,INCLUDE_DIRECTORIES>)
    target_compile_definitions(pf_preview PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
  endif()
endif()
